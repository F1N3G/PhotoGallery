```python
from pathlib import Path
import sqlite3
from datetime import datetime

from flask import (
    Flask, render_template, request, redirect,
    url_for, send_from_directory, flash
)

# Cai de baza ale proiectului
BASE_DIR = Path(__file__).resolve().parent
DB_PATH = BASE_DIR / "gallery.db"
UPLOAD_DIR = BASE_DIR / "uploads"

# Extensii acceptate la upload
ALLOWED_EXT = {".jpg", ".jpeg", ".png", ".webp", ".gif"}

# Initializare aplicatie Flask
app = Flask(__name__)
app.config["SECRET_KEY"] = "dev-secret"
UPLOAD_DIR.mkdir(exist_ok=True)


# Conectare la baza de date SQLite
def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


# Creeaza tabela si indexurile daca nu exista
def init_db():
    with get_db() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS photos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                filename TEXT NOT NULL,
                original_name TEXT NOT NULL,
                title TEXT NOT NULL,
                album TEXT NOT NULL DEFAULT 'General',
                tags TEXT NOT NULL DEFAULT '',
                size_bytes INTEGER NOT NULL,
                uploaded_at TEXT NOT NULL
            )
        """)
        conn.execute("CREATE INDEX IF NOT EXISTS idx_photos_uploaded_at ON photos(uploaded_at)")
        conn.execute("CREATE INDEX IF NOT EXISTS idx_photos_album ON photos(album)")


# Redirect din root catre galerie
@app.get("/")
def index():
    return redirect(url_for("gallery"))


# Afisarea galeriei
@app.get("/gallery")
def gallery():
    with get_db() as conn:
        photos = conn.execute(
            "SELECT * FROM photos ORDER BY uploaded_at DESC"
        ).fetchall()
    return render_template("gallery.html", photos=photos)


# Formularul de upload
@app.get("/upload")
def upload_form():
    return render_template("upload.html")


# Procesarea upload-ului
@app.post("/upload")
def upload_post():
    file = request.files.get("photo")
    title = (request.form.get("title") or "").strip()
    album = (request.form.get("album") or "General").strip() or "General"
    tags = (request.form.get("tags") or "").strip()

    if not file or not file.filename:
        flash("Selecteaza o poza.")
        return redirect(url_for("upload_form"))

    ext = Path(file.filename).suffix.lower()
    if ext not in ALLOWED_EXT:
        flash("Format invalid.")
        return redirect(url_for("upload_form"))

    if not title:
        title = Path(file.filename).stem

    # Nume unic pentru fisier
    stamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    filename = f"{stamp}{ext}"
    path = UPLOAD_DIR / filename
    file.save(path)

    uploaded_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    # Salvare metadate in DB
    with get_db() as conn:
        conn.execute("""
            INSERT INTO photos
            (filename, original_name, title, album, tags, size_bytes, uploaded_at)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (
            filename,
            file.filename,
            title,
            album,
            tags,
            path.stat().st_size,
            uploaded_at
        ))

    flash("Poza incarcata cu succes!")
    return redirect(url_for("gallery"))


# Stergerea unei poze
@app.post("/photo/<int:photo_id>/delete")
def delete_photo(photo_id):
    with get_db() as conn:
        photo = conn.execute(
            "SELECT * FROM photos WHERE id = ?",
            (photo_id,)
        ).fetchone()

        if not photo:
            flash("Poza nu exista.")
            return redirect(url_for("gallery"))

        conn.execute("DELETE FROM photos WHERE id = ?", (photo_id,))

    # Stergere fisier de pe disc
    file_path = UPLOAD_DIR / photo["filename"]
    if file_path.exists():
        file_path.unlink()

    flash("Poza a fost stearsa.")
    return redirect(url_for("gallery"))


# Servirea imaginilor din uploads
@app.get("/uploads/<path:filename>")
def uploads(filename):
    return send_from_directory(UPLOAD_DIR, filename)


# Pornirea aplicatiei
if __name__ == "__main__":
    init_db()
    app.run(debug=True)
```
